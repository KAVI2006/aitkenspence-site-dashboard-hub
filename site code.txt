### What was added for the transitions:
1. **Interactive Click Shockwave**: Clicking any option emits an expanding radial energy wave directly from your mouse coordinates across the hub canvas.
2. **Card Focus & Departure**: The clicked option button reacts with a spring-press and highlight glow, while neighboring cards gently fade and soften (`blur(3px)`).
3. **Direction-Aware 3D Depth Transitions**:
   - **Diving in (Forward)**: Entering Group or Sector dashboards smoothly zooms in from 3D space (`perspective(1000px) translateZ(40px)`).
   - **Going Back (Backward)**: Clicking `← Back` or ascending levels pulls back smoothly into view.
   - **Lateral (Pill Nav)**: Switching between tabs slides across with directional inertia.
4. **3D Spring Card Cascade**: New cards and category tiles cascade into view with a 3D perspective tilt and spring bounce (`rotateX(8deg)` to `0deg`).
5. **Debounce Protection**: Navigation is protected against accidental double-clicks during active transitions.

    function spawnShockwave(x, y) {

    function go(s, id, sub, ev) {
      if (isTransitioning) return;
      if (s === screen && id === sectorId && sub === subId) return;

      var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      var fromLevel = LEVEL[screen] != null ? LEVEL[screen] : 0;
      var toLevel = LEVEL[s] != null ? LEVEL[s] : 0;
      var isForward = toLevel > fromLevel;
      var isBack = toLevel < fromLevel;

      var hub = document.getElementById('pbihub');
      var hubRect = hub ? hub.getBoundingClientRect() : {left: 0, top: 0, width: 800, height: 600};
      var clickX = (ev && ev.clientX != null) ? ev.clientX : (hubRect.left + hubRect.width / 2);
      var clickY = (ev && ev.clientY != null) ? ev.clientY : (hubRect.top + hubRect.height / 3);

      if (!reduce) {
        spawnShockwave(clickX, clickY);
        if (ev && ev.target) {
          var btn = ev.target.closest ? ev.target.closest('.ph-bigbtn') : null;
          if (btn) {
            btn.classList.add('ph-chosen');
            var parentRow = btn.closest('.ph-big');
            if (parentRow) parentRow.classList.add('ph-leaving');
          }
        }
      }

      var dir = isForward ? 1 : (isBack ? -1 : ((s === 'sectors' && screen === 'group') ? 1 : -1));
      main.style.setProperty('--dx', (dir * 44) + 'px');

      var leavingClass = isForward ? 'leaving-forward' : (isBack ? 'leaving-back' : 'leaving-lateral');
      var enteringClass = isForward ? 'entering-forward' : (isBack ? 'entering-back' : 'entering-lateral');

      clearTimeout(goTimer);
      main.classList.remove('entering', 'entering-forward', 'entering-back', 'entering-lateral', 'leaving', 'leaving-forward', 'leaving-back', 'leaving-lateral');

      if (reduce) {
        screen = s; sectorId = id || null; subId = sub || null;
        render();
        return;
      }

      isTransitioning = true;
      main.classList.add(leavingClass);

      goTimer = setTimeout(function () {
        screen = s; sectorId = id || null; subId = sub || null;
        render();
        main.classList.remove(leavingClass);
        void main.offsetWidth;
        main.classList.add(enteringClass);

        setTimeout(function () {
          main.classList.remove(enteringClass);
          isTransitioning = false;
        }, 500);
      }, 250);
    }
